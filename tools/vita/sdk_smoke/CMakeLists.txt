set(OPENQ4_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
execute_process(
  COMMAND python3
    "${OPENQ4_ROOT}/tools/build/openq4_version.py"
    --source-root "${OPENQ4_ROOT}"
    --track dev
    --header-out "${CMAKE_CURRENT_BINARY_DIR}/openq4_version_generated.h"
  RESULT_VARIABLE OPENQ4_VERSION_RESULT
)
if(NOT OPENQ4_VERSION_RESULT EQUAL 0)
  message(FATAL_ERROR "Failed to generate openq4_version_generated.h")
endif()

# Compile-only integrity metadata for the Vita filesystem probe. The production
# pack hashes are generated by Meson from the real baseoq4 PK4s; this header
# exists only so FileSystem.cpp can be cross-compiled before packaging content.

"#ifndef OPENQ4_PAKS_GENERATED_H\n"
"#define OPENQ4_PAKS_GENERATED_H\n"
"#define OPENQ4_PAK0_MD5 \"00000000000000000000000000000000\"\n"
"#define OPENQ4_PAK0_SIZE_BYTES 0\n"
"#define OPENQ4_PAK0_FILE_COUNT 0\n"
"#define OPENQ4_PAK1_MD5 \"00000000000000000000000000000000\"\n"
"#define OPENQ4_PAK1_SIZE_BYTES 0\n"
"#define OPENQ4_PAK1_FILE_COUNT 0\n"
"#endif\n")
